```typescript
const ResultsPanel: React.FC<{
  calculationResult: CalculationResult;
}> = ({ calculationResult }) => {
  const {
    target_bonus,
    weighted_performance,
    pre_raf_bonus,
    initial_bonus,
    base_salary_cap,
    mrt_cap,
    final_bonus,
    cap_applied
  } = calculationResult;
  
  return (
    <div className="p-4 border rounded-lg">
      <h2 className="text-xl font-bold mb-4">Calculation Results</h2>
      
      {/* Final Bonus Display */}
      <div className="mb-6 p-4 bg-blue-50 rounded-lg text-center">
        <h3 className="text-lg font-bold mb-1">Final Bonus</h3>
        <p className="text-3xl font-bold text-blue-700">£{final_bonus.toLocaleString(undefined, { maximumFractionDigits: 0 })}</p>
        
        {cap_applied && (
          <div className="mt-2 p-2 bg-yellow-100 rounded text-sm text-yellow-800">
            {cap_applied === '3x_base' ? (
              <p>Capped at 3× Base Salary (£{base_salary_cap.toLocaleString()})</p>
            ) : cap_applied === 'mrt' ? (
              <p>MRT Cap Applied (£{mrt_cap.toLocaleString()})</p>
            ) : null}
          </div>
        )}
      </div>
      
      {/* Calculation Breakdown */}
      <div className="mb-4">
        <h3 className="font-bold mb-2">Calculation Breakdown</h3>
        
        <table className="w-full border-collapse">
          <tbody>
            <tr className="border-b">
              <td className="py-2 text-sm">Target Bonus</td>
              <td className="py-2 text-right font-mono">£{target_bonus.toLocaleString(undefined, { maximumFractionDigits: 0 })}</td>
            </tr>
            <tr className="border-b">
              <td className="py-2 text-sm">Weighted Performance</td>
              <td className="py-2 text-right font-mono">{weighted_performance.toFixed(2)}×</td>
            </tr>
            <tr className="border-b">
              <td className="py-2 text-sm">Pre-RAF Bonus</td>
              <td className="py-2 text-right font-mono">£{pre_raf_bonus.toLocaleString(undefined, { maximumFractionDigits: 0 })}</td>
            </tr>
            <tr className="border-b">
              <td className="py-2 text-sm">RAF Applied</td>
              <td className="py-2 text-right font-mono">{(initial_bonus / pre_raf_bonus).toFixed(2)}×</td>
            </tr>
            <tr className="border-b">
              <td className="py-2 text-sm">Initial Bonus</td>
              <td className="py-2 text-right font-mono">£{initial_bonus.toLocaleString(undefined, { maximumFractionDigits: 0 })}</td>
            </tr>
            {cap_applied && (
              <tr className="border-b bg-yellow-50">
                <td className="py-2 text-sm font-bold">Cap Applied</td>
                <td className="py-2 text-right font-mono font-bold">
                  {cap_applied === '3x_base' ? '3× Base Salary' : 'MRT Cap'}
                </td>
              </tr>
            )}
            <tr className="border-b font-bold">
              <td className="py-2">Final Bonus</td>
              <td className="py-2 text-right font-mono">£{final_bonus.toLocaleString(undefined, { maximumFractionDigits: 0 })}</td>
            </tr>
          </tbody>
        </table>
      </div>
      
      {/* Export Button */}
      <div className="mt-4">
        <button 
          onClick={() => {
            // Export logic here
            const csvContent = `Target Bonus,£${target_bonus}\n` +
                            `Weighted Performance,${weighted_performance}\n` +
                            `Pre-RAF Bonus,£${pre_raf_bonus}\n` +
                            `RAF Applied,${(initial_bonus / pre_raf_bonus).toFixed(2)}\n` +
                            `Initial Bonus,£${initial_bonus}\n` +
                            `${cap_applied ? 'Cap Applied,' + (cap_applied === '3x_base' ? '3× Base Salary' : 'MRT Cap') + '\n' : ''}` +
                            `Final Bonus,£${final_bonus}`;
            
            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            const url = URL.createObjectURL(blob);
            link.setAttribute('href', url);
            link.setAttribute('download', 'bonus_calculation.csv');
            link.style.visibility = 'hidden';
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
          }}
          className="px-4 py-2 bg-green-500 text-white rounded"
        >
          Export Results
        </button>
      </div>
    </div>
  );
};
```
