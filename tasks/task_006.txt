Example parameter configuration component:
```typescript
const BatchParameterConfig: React.FC<{
  onSubmit: (parameters: BatchParameters) => void;
  initialValues?: BatchParameters;
}> = ({ onSubmit, initialValues }) => {
  const [parameters, setParameters] = useState<BatchParameters>(initialValues || {
    targetBonusPct: 0.4, // 40%
    investmentWeight: 0.6, // 60%
    qualitativeWeight: 0.4, // 40%
    rafSensitivity: 0.2,
    rafLowerClamp: 0,
    rafUpperClamp: 1.2,
    mrtCapPct: 1.0, // 100% of base salary
    useDirectRaf: true, // Use RAF from CSV if available
  });
  
  const handleChange = (field: keyof BatchParameters, value: number | boolean) => {
    setParameters(prev => ({
      ...prev,
      [field]: value
    }));
    
    // If investment weight changes, adjust qualitative weight to maintain sum of 1
    if (field === 'investmentWeight') {
      setParameters(prev => ({
        ...prev,
        qualitativeWeight: 1 - Number(value)
      }));
    } else if (field === 'qualitativeWeight') {
      setParameters(prev => ({
        ...prev,
        investmentWeight: 1 - Number(value)
      }));
    }
  };
  
  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    onSubmit(parameters);
  };
  
  return (
    <form onSubmit={handleSubmit} className="p-4 border rounded-lg">
      <h2 className="text-xl font-bold mb-4">Global Calculation Parameters</h2>
      
      <div className="mb-4">
        <label className="block mb-2">Default Target Bonus %</label>
        <input
          type="number"
          min="0"
          max="2"
          step="0.01"
          value={parameters.targetBonusPct}
          onChange={(e) => handleChange('targetBonusPct', Number(e.target.value))}
          className="p-2 border rounded w-full"
        />
      </div>
      
      <div className="mb-4">
        <label className="block mb-2">Investment Weight</label>
        <input
          type="range"
          min="0"
          max="1"
          step="0.01"
          value={parameters.investmentWeight}
          onChange={(e) => handleChange('investmentWeight', Number(e.target.value))}
          className="w-full"
        />
        <div className="flex justify-between">
          <span>0%</span>
          <span>{(parameters.investmentWeight * 100).toFixed(0)}%</span>
          <span>100%</span>
        </div>
      </div>
      
      <div className="mb-4">
        <label className="block mb-2">Qualitative Weight</label>
        <input
          type="range"
          min="0"
          max="1"
          step="0.01"
          value={parameters.qualitativeWeight}
          onChange={(e) => handleChange('qualitativeWeight', Number(e.target.value))}
          className="w-full"
        />
        <div className="flex justify-between">
          <span>0%</span>
          <span>{(parameters.qualitativeWeight * 100).toFixed(0)}%</span>
          <span>100%</span>
        </div>
      </div>
      
      {/* Additional parameter inputs... */}
      
      <button 
        type="submit"
        className="px-4 py-2 bg-blue-500 text-white rounded"
      >
        Apply Parameters
      </button>
    </form>
  );
};
```
