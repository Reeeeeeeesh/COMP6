Example scenario parameter panel component:
```typescript
const ScenarioParameterPanel: React.FC<{
  initialParameters: ScenarioParameters;
  onChange: (parameters: ScenarioParameters) => void;
}> = ({ initialParameters, onChange }) => {
  const [parameters, setParameters] = useState<ScenarioParameters>(initialParameters);
  
  // Update parent component when parameters change
  useEffect(() => {
    onChange(parameters);
  }, [parameters, onChange]);
  
  const handleChange = (field: keyof ScenarioParameters, value: number | boolean) => {
    setParameters(prev => {
      const updated = {
        ...prev,
        [field]: value
      };
      
      // If investment weight changes, adjust qualitative weight to maintain sum of 1
      if (field === 'investmentWeight') {
        updated.qualitativeWeight = 1 - Number(value);
      } else if (field === 'qualitativeWeight') {
        updated.investmentWeight = 1 - Number(value);
      }
      
      return updated;
    });
  };
  
  return (
    <div className="p-4 border rounded-lg">
      <h2 className="text-xl font-bold mb-4">Scenario Parameters</h2>
      
      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
        {/* Performance Weights Section */}
        <div className="mb-4">
          <h3 className="font-bold mb-2">Performance Weights</h3>
          
          <div className="mb-2">
            <label className="block text-sm mb-1">Investment Weight</label>
            <input
              type="range"
              min="0"
              max="1"
              step="0.01"
              value={parameters.investmentWeight}
              onChange={(e) => handleChange('investmentWeight', Number(e.target.value))}
              className="w-full"
            />
            <div className="flex justify-between text-xs">
              <span>0%</span>
              <span>{(parameters.investmentWeight * 100).toFixed(0)}%</span>
              <span>100%</span>
            </div>
          </div>
          
          <div className="mb-2">
            <label className="block text-sm mb-1">Qualitative Weight</label>
            <input
              type="range"
              min="0"
              max="1"
              step="0.01"
              value={parameters.qualitativeWeight}
              onChange={(e) => handleChange('qualitativeWeight', Number(e.target.value))}
              className="w-full"
            />
            <div className="flex justify-between text-xs">
              <span>0%</span>
              <span>{(parameters.qualitativeWeight * 100).toFixed(0)}%</span>
              <span>100%</span>
            </div>
          </div>
        </div>
        
        {/* RAF Settings Section */}
        <div className="mb-4">
          <h3 className="font-bold mb-2">RAF Settings</h3>
          
          <div className="mb-2">
            <label className="block text-sm mb-1">RAF Sensitivity Factor</label>
            <input
              type="number"
              min="0"
              step="0.01"
              value={parameters.rafSensitivity}
              onChange={(e) => handleChange('rafSensitivity', Number(e.target.value))}
              className="p-2 border rounded w-full"
            />
          </div>
          
          <div className="mb-2">
            <label className="block text-sm mb-1">RAF Lower Clamp</label>
            <input
              type="number"
              min="0"
              step="0.01"
              value={parameters.rafLowerClamp}
              onChange={(e) => handleChange('rafLowerClamp', Number(e.target.value))}
              className="p-2 border rounded w-full"
            />
          </div>
          
          <div className="mb-2">
            <label className="block text-sm mb-1">RAF Upper Clamp</label>
            <input
              type="number"
              min="0"
              step="0.01"
              value={parameters.rafUpperClamp}
              onChange={(e) => handleChange('rafUpperClamp', Number(e.target.value))}
              className="p-2 border rounded w-full"
            />
          </div>
        </div>
        
        {/* Cap Settings Section */}
        <div className="mb-4">
          <h3 className="font-bold mb-2">Cap Settings</h3>
          
          <div className="mb-2">
            <label className="block text-sm mb-1">MRT Cap %</label>
            <input
              type="number"
              min="0"
              step="0.01"
              value={parameters.mrtCapPct}
              onChange={(e) => handleChange('mrtCapPct', Number(e.target.value))}
              className="p-2 border rounded w-full"
            />
          </div>
        </div>
      </div>
    </div>
  );
};
```
