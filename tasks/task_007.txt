```python
class BatchCalculationService:
    def __init__(self, db: Session):
        self.db = db
        self.calculation_engine = CalculationEngine()
    
    async def calculate_batch(self, batch_upload_id: int, global_parameters: dict) -> BatchCalculationResult:
        """Calculate bonuses for all employees in a batch."""
        # Get all employee data for this batch
        employee_data = await EmployeeDataDAL(self.db).get_by_batch_upload_id(batch_upload_id)
        
        # Create a new batch scenario for this calculation
        scenario = await BatchScenarioDAL(self.db).create(
            session_id=employee_data[0].batch_upload.session_id if employee_data else None,
            name=f"Batch {batch_upload_id} - {datetime.now().strftime('%Y-%m-%d %H:%M')}",
            global_parameters=global_parameters,
            is_saved=False
        )
        
        # Create a new batch calculation result
        batch_result = await BatchCalculationResultDAL(self.db).create(
            scenario_id=scenario.id,
            total_employees=len(employee_data),
            total_bonus_pool=0,  # Will update after calculations
            average_bonus=0,     # Will update after calculations
            capped_employees=0   # Will update after calculations
        )
        
        # Calculate bonus for each employee
        employee_results = []
        total_bonus = 0
        capped_count = 0
        
        for employee in employee_data:
            # Apply global parameters with potential individual overrides
            params = self._merge_parameters(global_parameters, employee.parameter_overrides or {})
            
            # Calculate bonus
            result = self.calculation_engine.calculate_final_bonus(
                base_salary=employee.base_salary,
                target_bonus_pct=employee.target_bonus_pct or params.get('targetBonusPct', 0),
                investment_weight=employee.investment_weight or params.get('investmentWeight', 0.6),
                investment_score_multiplier=employee.investment_score_multiplier,
                qualitative_weight=employee.qualitative_weight or params.get('qualitativeWeight', 0.4),
                qual_score_multiplier=employee.qual_score_multiplier,
                raf=employee.raf,
                is_mrt=employee.is_mrt,
                mrt_cap_pct=employee.mrt_cap_pct or params.get('mrtCapPct')
            )
            
            # Create employee calculation result
            employee_result = await EmployeeCalculationResultDAL(self.db).create(
                batch_result_id=batch_result.id,
                employee_data_id=employee.id,
                final_bonus=result['final_bonus'],
                pre_raf_bonus=result['pre_raf_bonus'],
                initial_bonus=result['initial_bonus'],
                applied_cap=result['cap_applied']
            )
            
            employee_results.append(employee_result)
            total_bonus += result['final_bonus']
            if result['cap_applied']:
                capped_count += 1
        
        # Update batch result with summary statistics
        average_bonus = total_bonus / len(employee_data) if employee_data else 0
        await BatchCalculationResultDAL(self.db).update(
            batch_result.id,
            total_bonus_pool=total_bonus,
            average_bonus=average_bonus,
            capped_employees=capped_count
        )
        
        return batch_result
    
    def _merge_parameters(self, global_params: dict, individual_overrides: dict) -> dict:
        """Merge global parameters with individual overrides."""
        result = global_params.copy()
        result.update(individual_overrides)
        return result
```
