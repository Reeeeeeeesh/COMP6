Example calculation function:
```python
def calculate_final_bonus(base_salary: float, target_bonus_pct: float, 
                         investment_weight: float, investment_score_multiplier: float,
                         qualitative_weight: float, qual_score_multiplier: float,
                         raf: float, is_mrt: bool = False, mrt_cap_pct: float = None) -> dict:
    """Calculate the final bonus with all intermediate steps."""
    # Calculate target bonus
    target_bonus = base_salary * target_bonus_pct
    
    # Calculate weighted performance
    weighted_performance = (investment_weight * investment_score_multiplier + 
                           qualitative_weight * qual_score_multiplier)
    
    # Calculate pre-RAF bonus
    pre_raf_bonus = target_bonus * weighted_performance
    
    # Apply RAF
    initial_bonus = pre_raf_bonus * raf
    
    # Apply caps
    base_salary_cap = base_salary * 3
    mrt_cap = base_salary * mrt_cap_pct if is_mrt and mrt_cap_pct else float('inf')
    
    # Apply the lower of the caps if either is breached
    final_bonus = min(initial_bonus, base_salary_cap, mrt_cap)
    
    return {
        "target_bonus": target_bonus,
        "weighted_performance": weighted_performance,
        "pre_raf_bonus": pre_raf_bonus,
        "initial_bonus": initial_bonus,
        "base_salary_cap": base_salary_cap,
        "mrt_cap": mrt_cap if is_mrt else None,
        "final_bonus": final_bonus,
        "cap_applied": "3x_base" if final_bonus == base_salary_cap and final_bonus < initial_bonus else
                      "mrt" if final_bonus == mrt_cap and final_bonus < initial_bonus else None
    }
```
