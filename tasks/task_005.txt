Example column mapping component:
```typescript
const ColumnMapper: React.FC<{
  sourceColumns: string[];
  targetFields: {name: string; required: boolean}[];
  onSubmit: (mappings: Record<string, string>, defaults: Record<string, any>) => void;
}> = ({ sourceColumns, targetFields, onSubmit }) => {
  const [mappings, setMappings] = useState<Record<string, string>>({});
  const [defaults, setDefaults] = useState<Record<string, any>>({});
  
  // Auto-mapping on component mount
  useEffect(() => {
    const suggestedMappings = {};
    
    targetFields.forEach(target => {
      // Find best match in source columns
      const bestMatch = sourceColumns.find(src => 
        src.toLowerCase().includes(target.name.toLowerCase()) ||
        target.name.toLowerCase().includes(src.toLowerCase())
      );
      
      if (bestMatch) {
        suggestedMappings[target.name] = bestMatch;
      }
    });
    
    setMappings(suggestedMappings);
  }, [sourceColumns, targetFields]);
  
  const handleMappingChange = (targetField: string, sourceColumn: string) => {
    setMappings(prev => ({
      ...prev,
      [targetField]: sourceColumn
    }));
  };
  
  const handleDefaultChange = (targetField: string, value: any) => {
    setDefaults(prev => ({
      ...prev,
      [targetField]: value
    }));
  };
  
  const handleSubmit = () => {
    onSubmit(mappings, defaults);
  };
  
  return (
    <div className="p-4">
      <h2 className="text-xl font-bold mb-4">Map Columns</h2>
      
      {targetFields.map(field => (
        <div key={field.name} className="mb-4">
          <label className="block mb-2">
            {field.name} {field.required && <span className="text-red-500">*</span>}
          </label>
          
          <select
            value={mappings[field.name] || ''}
            onChange={(e) => handleMappingChange(field.name, e.target.value)}
            className="p-2 border rounded w-full mb-2"
          >
            <option value="">-- Select source column --</option>
            {sourceColumns.map(col => (
              <option key={col} value={col}>{col}</option>
            ))}
          </select>
          
          {!mappings[field.name] && field.required && (
            <div>
              <label className="block mb-1 text-sm">Default value:</label>
              <input
                type="text"
                value={defaults[field.name] || ''}
                onChange={(e) => handleDefaultChange(field.name, e.target.value)}
                className="p-2 border rounded w-full"
                required
              />
            </div>
          )}
        </div>
      ))}
      
      <button 
        onClick={handleSubmit}
        className="px-4 py-2 bg-blue-500 text-white rounded"
      >
        Submit Mapping
      </button>
    </div>
  );
};
```
